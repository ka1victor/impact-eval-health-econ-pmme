\subsection{IVS e suas dimensões}
\rastreio{1. Motivação e Pergunta \textperiodcentered\ IVS e suas dimensões}

\begin{frame}{Quanto maior o IVS, mais difícil é exercer ali}
  \only<1>{%
    \pmmebuild{O que o índice mede}
    {\small \lead{IVS 2010, do Ipea} --- do \textit{Atlas da Vulnerabilidade
    Social nos Municípios Brasileiros}. Resume \num{16 indicadores} do
    \textbf{Censo 2010} em um número de \textbf{0 a 1}: quanto maior, mais
    vulnerável. Existe para \textbf{todos} os municípios, e é a \textbf{parte
    pública} da regra do valor --- a outra, a do Anexo IV, não é.}

    \vspace{0.9em}
    {\small As categorias do item 11.1.4 são as do próprio Atlas: \textbf{muito
    alta} acima de \num{0,500}, \textbf{alta} entre \num{0,400} e \num{0,500},
    e \textbf{demais} abaixo disso.}
  }%
  \only<2>{%
    \pmmebuild{As três dimensões}
    {\footnotesize\renewcommand{\arraystretch}{1.2}
    \begin{tabularx}{\textwidth}{@{}>{\bfseries\raggedright\arraybackslash}p{0.17\textwidth} >{\raggedright\arraybackslash}p{0.33\textwidth} L@{}}
      \toprule
      \normalfont\bfseries Dimensão & \bfseries Indicadores &
        \bfseries O que significa para quem vai atender ali \\
      \midrule
      Infraestrutura urbana & saneamento, coleta de lixo, tempo de
        deslocamento & morar e circular custam mais \\
      Capital humano & mortalidade infantil, analfabetismo, mães adolescentes &
        população mais doente, e serviço mais precário \\
      Renda e trabalho & extrema pobreza, desemprego, informalidade & quase
        não há mercado privado para complementar a renda \\
      \bottomrule
    \end{tabularx}}

    \vspace{0.7em}
    \begin{destaque}[0.93\textwidth]
      As três apontam para o mesmo lado: quanto maior o índice, mais caro é
      viver ali e mais duro é atender ali.
    \end{destaque}
  }%

  \fonte{\textbf{Fontes:} Ipea, \textit{Atlas da Vulnerabilidade Social nos
  Municípios Brasileiros} (2015) --- 16 indicadores do Censo 2010 em três
  sub-índices, e as faixas de classificação; Edital SGTES/MS nº~3/2025, item
  11.1.4; \texttt{modelo\_micro.md}, §3.1.}
\end{frame}

\subsection{Efeitos}
\rastreio{1. Motivação e Pergunta \textperiodcentered\ Efeitos}

\begin{frame}{O programa já deu sinais; a literatura aponta para os dois lados}
  \only<1>{%
    \pmmebuild{O que o ciclo 1 mostra}
    \begin{columns}[T]
      \begin{column}{0.495\textwidth}
        \includegraphics[width=\textwidth,height=0.42\textheight,keepaspectratio]%
          {preenchimento_ciclo1}
      \end{column}
      \begin{column}{0.495\textwidth}
        \includegraphics[width=\textwidth,height=0.42\textheight,keepaspectratio]%
          {oferta_total_mensal}
      \end{column}
    \end{columns}

    \vspace{0.6em}
    {\small Das \num{1.295 células} da primeira chamada, \num{393 (30,3\%)}
    tiveram alguém confirmado ou homologado. Nos \num{295 municípios} com curso
    de correspondência unívoca, o estoque de especialistas passou de \num{14,5}
    para \num{17,7} por 100 mil habitantes entre jun/2024 e jul/2026.}
  }%
  \only<2>{%
    \pmmebuild{A leitura}
    {\small
    \begin{itemize}
      \setlength{\itemsep}{0.5em}
      \item \lead{Por faixa, não há ordem.} Pagar o dobro (31,6\%) não
        preencheu mais que pagar uma vez e meia (37,4\%).
      \item \lead{Por território, há.} De \num{44,9\%} no metropolitano a
        \num{20,5\%} no interior remoto.
      \item \lead{O estoque sobe desde antes da oferta.} A alta de
        \num{22,5\%} começa em jun/2024, mais de um ano antes da publicação
        das vagas, e nenhum município da série está fora do programa.
    \end{itemize}}

    \vspace{0.9em}
    \begin{destaque}[0.93\textwidth]
      \textbf{Isto é descrição, não efeito.} As faixas diferem em muito mais
      que na bolsa, território prevê melhor que ela, e a série não tem grupo
      de comparação. Linguagem correta: \textbf{gradiente} e
      \textbf{associação}.
    \end{destaque}
  }%
  \only<3>{%
    \pmmebuild{A literatura aponta para os dois lados}
    \vspace{-0.9em}
    \begin{pmmeduas}
      \begin{pmmeparte}{Pagar mais funciona}
        \footnotesize\linespread{0.92}\selectfont
        \textbf{México, salário sorteado.} Em 106 postos, salário \num{33\%
        maior} elevou a aceitação em \num{15 p.p.}; a mais de 200 km da cidade
        natal, de 25\% para cerca de \num{80\%}, sem perda de qualificação.

        \vspace{0.15em}
        \textbf{O degrau tem o tamanho que a literatura pede.} O prêmio exigido
        para um posto pior vai de \num{37\% a 64\%} da renda anual, e a
        elasticidade da oferta no interior é \num{0,7}. O degrau do PMM-E é
        \num{+50\%}.
      \end{pmmeparte}
      \begin{pmmeparte}{Pagar mais não basta}
        \footnotesize\linespread{0.92}\selectfont
        \textbf{Brasil, o programa-irmão.} No Mais Médicos, \num{+15,1} médicos
        do programa por 100 mil viraram \num{+5,7} de expansão \textbf{líquida}.
        O resto substituiu quem já estava lá.

        \vspace{0.15em}
        \textbf{Austrália, a maioria não vai por preço.} De 3.727 clínicos,
        \num{65\%} ficaram onde estavam em \textbf{todos} os cenários. Para o
        pior posto, quem mudaria pedia \num{130\%} da renda anual.
      \end{pmmeparte}
    \end{pmmeduas}

    \vspace{-0.2em}
    {\footnotesize \textbf{Ressalva:} os percentuais são sobre a \textbf{renda
    total} do médico; a bolsa remunera \textbf{20 horas}.}

    \vspace{-0.1em}
    \begin{destaque}[0.93\textwidth]
      A evidência não decide se um degrau de R\$ 5 mil basta.
    \end{destaque}
  }%

  \fonte{\textbf{Fontes:} Dal Bó, Finan \& Rossi (2013), \textit{QJE}; Scott
  et al. (2013), \textit{Soc Sci Med} 96; Hone et al. (2020), \textit{BMC HSR}
  20:873. Ciclo 1: \texttt{output/tema\_trabalho/}, módulos A4 e A5. Série
  agregada: CNES 06/2024 a 07/2026 e Censo 2022 (IBGE); figuras por
  \texttt{scripts/apresentacao/gerar\_figuras\_banca1.py}.}

  % Duas colunas: a primeira reune "Trabalho" e "Entra no modelo como" do
  % documento (o trabalho em negrito, o papel em seguida); a segunda traz a
  % equacao original. Em tres colunas as celulas de texto quebravam em cinco
  % linhas e o frame nao cabia.
  {\scriptsize\renewcommand{\arraystretch}{1.0}
  \begin{tabularx}{\textwidth}{@{}L >{\centering\arraybackslash}p{0.3\textwidth}@{}}
    \toprule
    \bfseries Trabalho --- entra no modelo como & \bfseries Equações originais \\
    \midrule
    \textbf{Moehling, Niemesh, Thomasson \& Treber (2020)}, eq.~1, p.~184 ---
      escolha locacional intertemporal: o médico maximiza o valor presente do
      rendimento real, líquido de um custo não pecuniário &
      $\arg\max_{i \in I} \Bigl\{ \sum_t \delta^t \Bigl[
        \tfrac{\mathbb{E}(w_{it}^{(s)})}{p_{it}} - c_{it}^{(s)} \Bigr] \Bigr\}$ \\
    \addlinespace[0.15em]
    \textbf{Redding \& Rossi-Hansberg (2017)}, eq.~24, p.~28 --- equilíbrio
      espacial: amenidades e custo de moradia determinam a atratividade do
      lugar &
      $u_{nio} = \dfrac{z_{nio}\, B_n\, w_i}{\kappa_{ni}\, Q_n^{\,1-\beta}}$ \\
    \addlinespace[0.15em]
    \textbf{Choné \& Ma (2011)}, eq.~1, p.~232, com \textbf{Reinhardt (1972,
      1975)} --- utilidade do médico com altruísmo: atender cansa e satisfaz, e
      os dois passam por equipe e capital &
      $U = R - C(q; L, K) + \alpha B(q)$ \\
    \bottomrule
  \end{tabularx}}

  \vspace{0.15em}
  {\footnotesize Em Redding \& Rossi-Hansberg, o numerador é o que \lead{atrai}
  --- salário $w_i$, amenidades $B_n$, gosto pessoal $z_{nio}$ --- e o
  denominador é o que \lead{repele}: deslocamento $\kappa_{ni}$ e moradia
  $Q_n$.}

  \vspace{0em}
  \begin{destaque}[0.97\textwidth]
    \scriptsize \textbf{Atenção ao símbolo $B$.} Aqui $B_n$ é \textbf{amenidade} e $B(q)$
    é \textbf{benefício ao paciente}. A \textbf{bolsa} é $B_m$, e só aparece
    no slide 13.
  \end{destaque}

  \fonte{\textbf{Fontes:} Moehling et al. (2020), \textit{Cliometrica} 14,

  and Statistics} 101/102, p.~232, eq.~1; Reinhardt (1972, 1975);
  \texttt{modelo\_micro.md}, §1, §2 e §3.2.}
\end{frame}

\subsection{Modelo microeconômico conjunto}
\rastreio{2. Literatura e Modelo \textperiodcentered\ Modelo microeconômico conjunto}

\begin{frame}{A escolha locacional maximiza a renda real líquida}
  \only<1>{%
    \pmmebuild{A equação de escolha}
    {\small O médico $i$ escolhe o município $m$ que maximiza o valor presente
    do rendimento \lead{real}, líquido do custo não pecuniário de viver e
    atender ali:}

    \vspace{-0.5em}
    \[
      \arg\max_{m \in M} \left\{ \sum_t \delta^t
      \left[ \frac{\mathbb{E}(w_{mt}^{(s)})}{p_{mt}} - c_{im}^{(s)} \right] \right\}
    \]
    \vspace{-1.1em}

    {\footnotesize\renewcommand{\arraystretch}{1.05}
    \begin{tabularx}{\textwidth}{@{}>{\centering\arraybackslash}p{0.17\textwidth} L@{}}
      \toprule
      \bfseries Termo & \bfseries O que é \\
      \midrule
      $m \in M$ & \textbf{municípios} candidatos \\
      $\mathbb{E}(w^{(s)}_{mt})$ & remuneração esperada em $m$, no ano $t$, na
        especialidade $s$ \\
      $p_{mt}$ & nível de preços local, o deflator \\
      $c^{(s)}_{im}$ & custo \textbf{não pecuniário} de viver e atender ali \\
      $\sum_t \delta^t$ & a escolha é de \textbf{carreira}, não de um mês \\
      \bottomrule
    \end{tabularx}}
  }%
  \only<2>{%
    \pmmebuild{Interpretação}
    {\small Deflaciona a remuneração, subtrai o custo do lugar, desconta a
    carreira inteira pelo fator $\delta$ e devolve o município de maior
    valor.}

    \vspace{0.4em}
    \begin{destaque}[0.93\textwidth]
      Salário nominal alto não compensa preços e custos locais altos.
    \end{destaque}

    \vspace{0.4em}
    {\small Na equação original $i$ é a localidade; \lead{daqui em diante $i$ é
    o médico e $m$ é o município}, porque o custo depende de quem escolhe, e
    não só de onde.}

    \vspace{0.6em}
    {\small Para os próprios autores, $c$ reúne \textit{``preferences over
    rural or urban living \ldots\ such as proximity to family''} --- uma
    \lead{caixa-preta}. Os dois slides seguintes a abrem: primeiro o
    \textbf{custo} do lugar, depois a \textbf{remuneração} dele.}
  }%

  \fonte{\textbf{Fontes:} Moehling et al. (2020), \textit{Cliometrica} 14,
  p.~184, eq.~1; \texttt{modelo\_micro.md}, §1 e §2.4.}
\end{frame}

\subsection{Custo da localidade}
\rastreio{2. Literatura e Modelo \textperiodcentered\ Custo da localidade}

\begin{frame}{O custo da localidade tem duas metades: o lugar e o trabalho}
  \only<1>{%
    \pmmebuild{O lugar}
    {\small De Redding \& Rossi-Hansberg, o custo do \lead{lugar}:}

    \vspace{-0.3em}
    \[ c^{\text{geo}}_{im} = \phi(\text{dist}_{im}) - \gamma A_m + \theta_i^{\text{rural}} \]
    \vspace{-0.7em}

    {\small $\phi(\text{dist}_{im})$ é o afastamento da família, $A_m$ são as
    amenidades urbanas e $\theta_i^{\text{rural}}$ é o gosto pessoal por
    cidade pequena --- o único termo sem sinal universal.}
  }%
  \only<2>{%
    \pmmebuild{O trabalho}
    {\small De Choné \& Ma, com Reinhardt, o custo do \lead{trabalho}:}

    \vspace{-0.3em}
    \[ c^{\text{laboral}}_{im} = C(q; L_m, K_m) - \alpha_i B(q; L_m, K_m) \]
    \vspace{-0.7em}

    {\small $C$ é o cansaço de atender $q$ pacientes, $B$ é o benefício gerado
    a eles e $\alpha_i$ é o altruísmo do médico. Atender cansa de forma
    crescente ($C'' > 0$) e curar satisfaz de forma decrescente ($B'' < 0$): a
    curva do custo líquido é um \lead{U} --- atender mais compensa até um
    mínimo, e depois exaure.}

    \vspace{0.6em}
    {\small \lead{Extensão deste projeto}, motivada por Reinhardt: escrever
    $B(q; L, K)$, e não $B(q)$ --- equipe e capital não só poupam esforço como
    ampliam o que o atendimento produz.}
  }%
  \only<3>{%
    \pmmebuild{As três desvantagens, na visão do médico}
    \vspace{-0.2em}
    {\footnotesize\renewcommand{\arraystretch}{1.05}
    \begin{tabularx}{\textwidth}{@{}>{\bfseries\raggedright\arraybackslash}p{0.15\textwidth} >{\centering\arraybackslash}p{0.09\textwidth} L >{\raggedright\arraybackslash}p{0.15\textwidth}@{}}
      \toprule
      \normalfont\bfseries Desvantagem & \bfseries Termo & \bfseries Por que pesa &
        \bfseries Medimos? \\
      \midrule
      Sem retaguarda & $L_m$ & não há segunda opinião nem a quem encaminhar:
        cansa mais e resolve menos & sim, colegas no CNES \\
      Sem infraestrutura & $K_m$ & falta leito, insumo e equipamento: o mesmo
        esforço rende menos saúde & em parte \\
      Longe da família & $\phi(\text{dist})$ & cada quilômetro custa, e
        ninguém paga por ele & não \\
      \bottomrule
    \end{tabularx}}

    \vspace{0.3em}
    {\small No Brasil, entre 50 mil generalistas formados de 2001 a 2013,
    \lead{a proximidade do lugar de nascimento ou de formação é o principal
    fator}; salário e infraestrutura pesam menos.}

    \vspace{0.3em}
    {\footnotesize \textbf{Limitação assumida:} CNES e edital não informam a
    residência do profissional, por sigilo fiscal. A unidade é o
    \textbf{município do estabelecimento}; a distância entra como latente.}
  }%

  \fonte{\textbf{Fontes:} \texttt{modelo\_micro.md}, §2.1 a §2.3 e §3.2, de
  onde vêm as duas equações inferidas; Costa, Nunes \& Sanches (2024),
  \textit{REStat}; as equações originais estão no slide 10.}
\end{frame}

\subsection{Remuneração da localidade}
\rastreio{2. Literatura e Modelo \textperiodcentered\ Remuneração da localidade}

\begin{frame}{A bolsa é o piso da remuneração, não o total}
  \only<1>{%
    \pmmebuild{A remuneração total}
    {\small A lei fixa a \lead{bolsa}, não a remuneração. A bolsa compra
    \textbf{20 horas}; o que o médico ganha além delas é mercado local. Por
    construção, a remuneração é maior ou igual à bolsa:}

    \vspace{-0.6em}
    \[
      \mathbb{E}(w_{imt} \mid B_m)
      = \underbrace{B_m}_{\text{fixado pela regra}}
      + \underbrace{w^{\text{priv}}_m}_{\text{mercado local}}
      \;\geq\; B_m, \qquad w^{\text{priv}}_m \geq 0
    \]
    \vspace{-1.25em}

    {\footnotesize\renewcommand{\arraystretch}{1.0}
    \begin{tabularx}{\textwidth}{@{}>{\bfseries\raggedright\arraybackslash}p{0.17\textwidth} L L@{}}
      \toprule
      & \bfseries Capital ou metrópole & \bfseries Interior isolado \\
      \midrule
      Bolsa $B_m$ & R\$ 10 mil & R\$ 20 mil \\
      Mercado $w^{\text{priv}}_m$ & maior & \textbf{menor --- não nulo} \\
      Total $w$ & $\geq$ R\$ 10 mil, e pode passar dos R\$ 20 mil &
        $\geq$ R\$ 20 mil, e perto do piso \\
      Deflator $p_m$ & custo de vida alto & supomos menor, e não é garantido:
        o custo logístico encarece parte da cesta \\
      \bottomrule
    \end{tabularx}}
  }%
  \only<2>{%
    \pmmebuild{Interpretação}
    {\small A bolsa do interior é o \lead{dobro} da da capital, e ainda assim a
    \lead{remuneração total} pode ser \lead{menor} lá: o que a regra acrescenta,
    o mercado local deixa de acrescentar. O deflator puxa no sentido oposto, e
    não se sabe a priori qual força vence.}

    \vspace{0.15em}
    \begin{destaque}[0.93\textwidth]
      Esta é a quarta desvantagem do lugar: onde o mercado privado é fino, a
      remuneração encosta no piso da bolsa.
    \end{destaque}

    \vspace{0.15em}
    {\small Pela dupla prática do slide 4, $w^{\text{priv}} > 0$ é a regra, não
    a exceção --- o interior isolado tem \textbf{menos} mercado, não nenhum.}

    \vspace{0.3em}
    {\small A política aposta que R\$ 5 mil compensam o lugar. O modelo diz que
    eles competem com um mercado privado cuja escassez é, ela própria, uma
    desvantagem do lugar.}
  }%

  \fonte{\textbf{Fontes:} \texttt{modelo\_micro.md}, §3;
  \texttt{hipoteses\_e\_viabilidade\_empirica.md}, §2; Lei nº~15.233/2025,
  art.~21; Edital SGTES/MS nº~3/2025, itens 11.1.3 e 11.3.b.}
\end{frame}

\subsection{Implicações para o PMM-E}
\rastreio{3. Hipótese e Viabilidade \textperiodcentered\ Implicações para o PMM-E}

\begin{frame}{No PMM-E, a regra fixa a remuneração e o IVS organiza o custo}
  \only<1>{%
    \pmmebuild{O modelo integrado}
    \vspace{-1.3em}
    \[
      V_{im} = \sum\nolimits_t \delta^t \Bigl[
        \frac{\mathbb{E}(w_{imt} \mid \mathbf{B}_m)}{p_{mt}} - c_{im} \Bigr]
    \]
    \vspace{-1.1em}

    {\scriptsize para o médico $i$ no município $m$, com
    $\mathbb{E}(w \mid \mathbf{B}_m) = \mathbf{B}_m + \mathbf{w}^{\text{priv}}_m$
    do slide 13 e $c_{im} = c^{\text{geo}}_{im} + c^{\text{laboral}}_{im}$ do
    slide 12.}

    \vspace{0em}
    {\scriptsize\renewcommand{\arraystretch}{1.0}
    \begin{tabularx}{\textwidth}{@{}>{\centering\arraybackslash}p{0.06\textwidth} >{\raggedright\arraybackslash}p{0.24\textwidth} >{\centering\arraybackslash}p{0.25\textwidth} L@{}}
      \toprule
      \bfseries Termo & \bfseries Variável do programa & \bfseries Derivada & \bfseries Leitura \\
      \midrule
      $\mathbf{B}_m$ & valor da bolsa & $\partial V/\partial \mathbf{B}_m > 0$ &
        é o instrumento \\
      $\mathbf{w}^{\text{priv}}_m$ & mercado local, não observado &
        $\partial^2 V/\partial \mathbf{B}_m\, \partial \mathbf{w}^{\text{priv}} < 0$ &
        a bolsa vale mais onde há menos mercado \\
      $p_m$ & custo de vida, por UF & $\partial V/\partial p_m < 0$ & opera
        contra a vulnerabilidade \\
      $c_{im}$ & \textbf{IVS} e suas dimensões, do slide 6 & $c_0'(IVS) > 0$ &
        o custo \textbf{cresce} com o índice \\
      \bottomrule
    \end{tabularx}}

    \vspace{0em}
    \begin{destaque}[0.97\textwidth]
      \scriptsize \textbf{O que Moehling, Redding \& Rossi-Hansberg e Choné
      \& Ma não têm:} remuneração fixada por \textbf{regra pública sobre um
      índice territorial}. É só isso que a adaptação ao PMM-E acrescenta.
    \end{destaque}
  }%
  \only<2>{%
    \pmmebuild{A condição de aceitação}
    {\small Distância, aluguel, mercado local e esforço clínico \lead{não são
    observados}; o IVS é. O custo do lugar entra por ele, e o que é do médico
    fica no desvio individual: $c_{im} = c_0(IVS_m) + \eta_i$.}

    \vspace{0.5em}
    {\small O médico $i$ aceita a vaga em $m$ quando ela supera sua melhor
    alternativa $\bar{v}_i$:}

    \vspace{-0.3em}
    \[
      \frac{\mathbf{B}_m + \mathbf{w}^{\text{priv}}_m}{p_m} - c_0(IVS_m)
      \;\geq\; \bar{v}_i
    \]
    \vspace{-0.6em}

    {\small A vaga é preenchida se existir \lead{ao menos um} candidato para
    quem isso vale. Tudo que aumenta o lado esquerdo aumenta essa
    probabilidade.}
  }%
  \only<3>{%
    \pmmebuild{Na fronteira entre faixas}
    {\small \lead{O custo é obstáculo, não hipótese.} Ele não varia livremente:
    a regra do edital o amarra à bolsa, e os dois sobem juntos. Preencher a
    vaga do lado mais vulnerável da fronteira exige}

    \vspace{-0.2em}
    \[
      \frac{\Delta \mathbf{B}_m}{p_m} > \Delta c_0,
      \qquad \Delta \mathbf{B}_m = \text{R\$ } 5.000
    \]
    \vspace{-0.2em}

    \begin{destaque}[0.93\textwidth]
      A pergunta da apresentação é se essa desigualdade vale.
    \end{destaque}
  }%

  \fonte{\textbf{Fontes:} \texttt{modelo\_micro.md}, §2.4, §3, §3.1 e §4.1;
  \texttt{hipoteses\_e\_viabilidade\_empirica.md}, §3; Edital SGTES/MS
  nº~3/2025, item 11.1.3.}
\end{frame}
